# The existing native lifecycle, compiled against the exact reviewed production stage.
if(NOT EXISTS "${OUTLOUD_INSTALLER_PACKAGE}")
    message(FATAL_ERROR "Missing verified installer inventory")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${OUTLOUD_INSTALLER_PACKAGE}")
file(READ "${OUTLOUD_INSTALLER_PACKAGE}" package)
string(JSON count LENGTH "${package}" files)
if(count LESS 1 OR count GREATER 4096)
    message(FATAL_ERROR "Invalid installer inventory count")
endif()
math(EXPR last "${count}-1")
set(paths "#pragma once\ninline constexpr const wchar_t* registration_payload_paths[]={\n")
set(rows "#pragma once\nstruct MaintenancePackageRow {const wchar_t* path;bool pe;bool product_version;uint64_t bytes;const wchar_t* hash;bool sign;};\ninline constexpr MaintenancePackageRow maintenance_package[]={\n")
foreach(i RANGE ${last})
    string(JSON path GET "${package}" files ${i} path)
    if(NOT path MATCHES "^[A-Za-z0-9_./-]+$")
        message(FATAL_ERROR "Unsupported installer payload path: ${path}")
    endif()
    foreach(key pe product_version sign)
        string(JSON value GET "${package}" files ${i} ${key})
        if(value)
            set(${key} true)
        else()
            set(${key} false)
        endif()
    endforeach()
    string(JSON bytes GET "${package}" files ${i} bytes)
    string(JSON hash GET "${package}" files ${i} sha256)
    string(APPEND paths "L\"${path}\",\n")
    string(APPEND rows "{L\"${path}\",${pe},${product_version},${bytes},L\"${hash}\",${sign}},\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registration_payload_paths.hpp" "${paths}};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/maintenance_package.hpp" "${rows}};\n")

function(production_target target source)
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN OUTLOUD_MAINTENANCE_VERSION="${OUTLOUD_PRODUCT_VERSION}")
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive-)
    target_link_libraries(${target} PRIVATE advapi32 crypt32 ncrypt bcrypt wintrust shell32 ole32 ktmw32 rstrtmgr version)
endfunction()
