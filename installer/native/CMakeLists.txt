# The detector must run before the VC Redistributable or .NET is installed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${OUTLOUD_HOSTFXR_HEADERS}/hostfxr.h")
    message(FATAL_ERROR "Use the reviewed installed .NET apphost pack's hostfxr.h.")
endif()

target_compile_options(outloud-prerequisite-tests PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(outloud-runtime-install-tests runtime_install_tests.cpp)
target_compile_options(outloud-runtime-install-tests PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(OutLoudInstallSigning signing.cpp probe.manifest)
# Use the package's exact payload allowlist in native pre-registration validation;
# do not maintain a second independent list or accept arbitrary manifest paths.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../inno/package.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../inno/package.json" registration_package)
string(JSON registration_count LENGTH "${registration_package}" files)
math(EXPR registration_last "${registration_count}-1")
set(registration_header "#pragma once\ninline constexpr const wchar_t* registration_payload_paths[]={\n")
set(maintenance_header "#pragma once\nstruct MaintenancePackageRow {const wchar_t* path;bool pe;uint64_t bytes;const wchar_t* hash;};\ninline constexpr MaintenancePackageRow maintenance_package[]={\n")
foreach(registration_index RANGE ${registration_last})
    string(JSON registration_file GET "${registration_package}" files ${registration_index} path)
    if(NOT registration_file MATCHES "^[A-Za-z0-9_./-]+$")
        message(FATAL_ERROR "Unsupported registration payload path.")
    endif()
    string(APPEND registration_header "L\"${registration_file}\",\n")
    string(JSON maintenance_pe GET "${registration_package}" files ${registration_index} pe)
    if(maintenance_pe)
        set(maintenance_pe true)
    else()
        set(maintenance_pe false)
    endif()
    string(JSON maintenance_bytes ERROR_VARIABLE maintenance_error GET "${registration_package}" files ${registration_index} bytes)
    if(maintenance_error)
        set(maintenance_bytes 0)
        set(maintenance_hash "")
    else()
        string(JSON maintenance_hash GET "${registration_package}" files ${registration_index} sha256)
    endif()
    string(APPEND maintenance_header "{L\"${registration_file}\",${maintenance_pe},${maintenance_bytes},L\"${maintenance_hash}\"},\n")
endforeach()
string(APPEND registration_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registration_payload_paths.hpp" "${registration_header}")
string(APPEND maintenance_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/maintenance_package.hpp" "${maintenance_header}")
target_include_directories(OutLoudInstallSigning PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(OutLoudInstallSigning PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN OUTLOUD_MAINTENANCE_VERSION="${OUTLOUD_PRODUCT_VERSION}")
target_compile_options(OutLoudInstallSigning PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_libraries(OutLoudInstallSigning PRIVATE advapi32 crypt32 ncrypt bcrypt wintrust shell32 ole32 ktmw32 rstrtmgr version)
target_link_options(OutLoudInstallSigning PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(outloud-install-signing-tests signing_tests.cpp)
target_compile_options(outloud-install-signing-tests PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(outloud-install-registration-tests registration_tests.cpp)
target_compile_definitions(outloud-install-registration-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(outloud-install-registration-tests PRIVATE /W4 /WX /utf-8 /permissive-)
