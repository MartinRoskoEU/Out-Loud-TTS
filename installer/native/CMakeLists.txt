cmake_minimum_required(VERSION 3.24)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/OutLoudVersion.cmake")
outloud_read_product_version("${CMAKE_CURRENT_SOURCE_DIR}/../../ProductVersion.props")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../ProductVersion.props")
project(outloud_prerequisites VERSION "${OUTLOUD_PRODUCT_VERSION}" LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The deployment inspector targets Windows x64/MSVC.")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the inspector independent of deployed application/VC/.NET runtimes.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(OutLoudInstallSigning signing.cpp probe.manifest)
target_compile_definitions(OutLoudInstallSigning PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(OutLoudInstallSigning PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_libraries(OutLoudInstallSigning PRIVATE advapi32 crypt32 bcrypt wintrust shell32 ole32)
target_link_options(OutLoudInstallSigning PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
outloud_add_version_resource(OutLoudInstallSigning "Out Loud deployment inspection")

# Existing isolated native checker now covers the retained read-only boundaries.
add_executable(outloud-install-signing-tests signing_tests.cpp)
target_compile_definitions(outloud-install-signing-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(outloud-install-signing-tests PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_libraries(outloud-install-signing-tests PRIVATE advapi32 crypt32 bcrypt wintrust shell32 ole32)

# Explicit production staging leaves the development inspector and its inventory unchanged.
set(OUTLOUD_INSTALLER_PACKAGE "" CACHE FILEPATH "Verified self-contained installer package inventory")
if(OUTLOUD_INSTALLER_PACKAGE)
    add_subdirectory(production)
endif()
